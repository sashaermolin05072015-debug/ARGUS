// ================= ICON DRAWINGS =================
void drawIcon(int id, int x, int y, uint16_t color, int size) {
  // Compact line-art icons designed for the 160x128 ST7735 screen.
  if (size < 1) size = 1;
  switch (id) {
    case 0: { // Wi-Fi signal
      for (int arc = 0; arc < 3; ++arc) {
        int r = size * (4 + arc * 3);
        int lastX = 0, lastY = 0;
        bool haveLast = false;
        for (int deg = 225; deg <= 315; deg += 5) {
          float a = deg * PI / 180.0f;
          int px = x + (int)(cosf(a) * r);
          int py = y + (int)(sinf(a) * r);
          if (haveLast) tft.drawLine(lastX, lastY, px, py, color);
          lastX = px; lastY = py; haveLast = true;
        }
      }
      tft.fillCircle(x, y + size * 2, size, color);
      break;
    }
    case 1: { // IR remote with IR rays
      int w = size * 8, h = size * 15;
      tft.drawRoundRect(x - w/2, y - h/2, w, h, size * 2, color);
      tft.drawCircle(x, y - size*4, size, color);
      tft.fillCircle(x, y - size*4, max(1, size/2), color);
      tft.fillCircle(x, y, max(1, size/2), color);
      tft.fillCircle(x, y + size*3, max(1, size/2), color);
      // IR rays above the remote
      tft.drawLine(x - size*6, y - size*10, x - size*8, y - size*12, color);
      tft.drawLine(x, y - size*10, x, y - size*13, color);
      tft.drawLine(x + size*6, y - size*10, x + size*8, y - size*12, color);
      break;
    }
    case 2: { // Gear
      tft.drawCircle(x, y, size*6, color);
      tft.drawCircle(x, y, size*3, color);
      for (int i=0; i<8; ++i) {
        float a = i * PI / 4.0f;
        tft.drawLine(x + (int)(cosf(a)*size*6), y + (int)(sinf(a)*size*6),
                     x + (int)(cosf(a)*size*9), y + (int)(sinf(a)*size*9), color);
      }
      break;
    }
    case 3: { // Info
      tft.drawCircle(x, y, size*7, color);
      tft.fillCircle(x, y-size*3, max(1,size), color);
      tft.fillRoundRect(x-size, y-size, size*2, size*6, size, color);
      break;
    }
    case 4: { // Radar
      for (int r=size*3; r<=size*9; r+=size*3) tft.drawCircle(x,y,r,color);
      tft.drawFastHLine(x-size*9,y,size*18,color);
      tft.drawFastVLine(x,y-size*9,size*18,color);
      tft.drawLine(x,y,x+size*7,y-size*6,color);
      tft.fillCircle(x,y,size,color);
      break;
    }
    case 5: { // Tools / wrench
      tft.drawCircle(x-size*3,y-size*4,size*3,color);
      tft.drawCircle(x-size*3,y-size*4,size,color);
      tft.drawLine(x-size,y-size*2,x+size*6,y+size*6,color);
      tft.drawLine(x+size*3,y+size*6,x+size*7,y+size*2,color);
      tft.drawLine(x+size*4,y+size*7,x+size*7,y+size*4,color);
      break;
    }
    case 7: { // RGB LED symbol
      tft.drawCircle(x, y, size*6, color);
      tft.fillCircle(x-size*3, y, size, color);
      tft.fillCircle(x, y-size*3, size, color);
      tft.fillCircle(x+size*3, y, size, color);
      tft.fillCircle(x, y+size*3, size, color);
      break;
    }
    case 6: { // ARGUS: eye inside a targeting frame
      tft.drawRoundRect(x-size*11,y-size*8,size*22,size*16,size*3,color);
      tft.drawLine(x-size*8,y,x-size*3,y-size*4,color);
      tft.drawLine(x-size*3,y-size*4,x,y-size*5,color);
      tft.drawLine(x,y-size*5,x+size*3,y-size*4,color);
      tft.drawLine(x+size*3,y-size*4,x+size*8,y,color);
      tft.drawLine(x-size*8,y,x-size*3,y+size*4,color);
      tft.drawLine(x-size*3,y+size*4,x,y+size*5,color);
      tft.drawLine(x,y+size*5,x+size*3,y+size*4,color);
      tft.drawLine(x+size*3,y+size*4,x+size*8,y,color);
      tft.drawCircle(x,y,size*3,color);
      tft.fillCircle(x,y,size,color);
      tft.fillCircle(x+size,y-size, max(1,size/2), COL_TEXT);
      break;
    }
  }
}

// ================= BUZZER =================
void toneRaw(uint16_t freq, uint16_t dur) {
if (!buzzerEnabled || freq == 0) { delay(dur); return; }
ledcWriteTone(BUZZER_PIN, freq);
delay(dur);
ledcWriteTone(BUZZER_PIN, 0);
}
void beepNav()     { toneRaw(2000, 25); }
void beepSelect()  { toneRaw(2800, 50); }
void beepBack()    { toneRaw(1500, 40); }
void beepError()   { toneRaw(800, 120); }
void beepSuccess() { toneRaw(2500, 60); delay(20); toneRaw(3200, 60); }
void playBootMelody() {
const uint16_t n[] = {523,659,784,1047};
const uint16_t d[] = {100,100,100,300};
for (int i = 0; i < 4; i++) toneRaw(n[i], d[i]);
}
void playAttackAlarm() { toneRaw(3000,80); toneRaw(2000,80); toneRaw(3000,80); }
void playSnifferPing() { toneRaw(2400,25); }
void playIRCue()       { toneRaw(3000,40); }

void applyBrightness() {
if (brightnessLevel < 0) brightnessLevel = 0;
if (brightnessLevel >= BRIGHTNESS_LEVELS) brightnessLevel = BRIGHTNESS_LEVELS - 1;
ledcAttach(TFT_BL, 5000, 8);
ledcWrite(TFT_BL, brightnessValues[brightnessLevel]);
}

// ================= BUTTONS =================
bool backPressed() {
if (digitalRead(BTN_BACK) == LOW && millis() - lastButton[0] > DEBOUNCE_MS) {
lastButton[0] = millis(); beepBack(); return true;
}
return false;
}
bool nextPressed() {
if (digitalRead(BTN_NEXT) == LOW && millis() - lastButton[1] > DEBOUNCE_MS) {
lastButton[1] = millis(); beepNav(); return true;
}
return false;
}
bool selectPressed() {
if (digitalRead(BTN_SELECT) == LOW && millis() - lastButton[2] > DEBOUNCE_MS) {
lastButton[2] = millis(); beepSelect(); return true;
}
return false;
}

int w = strlen(s) * 6 * size;
int x = (160 - w) / 2; if (x < 0) x = 0;
text(s, x, y, c, size);
}
void truncateTo(const char* src, char* out, int maxChars) {
int len = strlen(src);
if (len <= maxChars) { strcpy(out, src); return; }
if (maxChars <= 2) { strncpy(out, src, maxChars); out[maxChars] = '\0'; return; }
strncpy(out, src, maxChars - 2);
out[maxChars - 2] = '.'; out[maxChars - 1] = '.'; out[maxChars] = '\0';
}

tft.fillRect(0, 0, 160, 22, COL_ACCENT);

}

void sendIRCommand(const IRCommand& c) {
uint32_t now = millis();
if ((uint32_t)(now - lastIrSend) < IR_SEND_GUARD_MS) return;
switch (c.protocol) {
case IR_PROTO_NEC:        irsend.sendNEC(c.code,c.bits,c.repeat); break;
case IR_PROTO_SAMSUNG:    irsend.sendSAMSUNG(c.code,c.bits,c.repeat); break;
case IR_PROTO_LG:         irsend.sendLG(c.code,c.bits,c.repeat); break;
case IR_PROTO_SONY:       irsend.sendSony(c.code,c.bits,c.repeat); break;
case IR_PROTO_PANASONIC:  irsend.sendPanasonic64(c.code,c.bits,c.repeat); break;
case IR_PROTO_SHARP:      irsend.sendSharpRaw(c.code,c.bits,c.repeat); break;
case IR_PROTO_RC5:        irsend.sendRC5(c.code,c.bits,c.repeat); break;
default: return;
}
lastIrSend = millis();
playIRCue();
}

// ================= CUTE ORANGE TETRIS (ARGUS APP) =================
namespace CuteTetris {
  const int W=10, H=10, CELL=10, X=30, Y=17;
  const uint16_t BG=0x10A2, PANEL=0x2124, ORANGE=0xFD20, PEACH=0xFE6A, CREAM=0xFFDE, GRID=0x39C7, GHOST=0x7BEF;
  uint8_t board[H][W];
  uint32_t score=0, lastFall=0, lastHorizontal=0, selectDownAt=0;
  uint16_t lines=0;
  uint8_t level=1, type=0, nextType=0, rot=0;
  int8_t px=3, py=0;
  bool over=false, selectWasDown=false, selectLongDone=false;
  bool backWasDown=false, nextWasDown=false, comboWasDown=false;
  int8_t bag[7]; uint8_t bagPos=7;
  const int8_t shape[7][4][4][2] = {
    {{{0,1},{1,1},{2,1},{3,1}},{{2,0},{2,1},{2,2},{2,3}},{{0,2},{1,2},{2,2},{3,2}},{{1,0},{1,1},{1,2},{1,3}}},
    {{{1,0},{2,0},{1,1},{2,1}},{{1,0},{2,0},{1,1},{2,1}},{{1,0},{2,0},{1,1},{2,1}},{{1,0},{2,0},{1,1},{2,1}}},
    {{{1,0},{0,1},{1,1},{2,1}},{{1,0},{1,1},{2,1},{1,2}},{{0,1},{1,1},{2,1},{1,2}},{{1,0},{0,1},{1,1},{1,2}}},
    {{{1,0},{2,0},{0,1},{1,1}},{{1,0},{1,1},{2,1},{2,2}},{{0,1},{1,1},{2,1},{3,1}},{{0,0},{0,1},{1,1},{1,2}}},
    {{{0,0},{1,0},{1,1},{2,1}},{{2,0},{1,1},{2,1},{1,2}},{{0,1},{1,1},{1,2},{2,2}},{{1,0},{0,1},{1,1},{0,2}}},
    {{{0,0},{0,1},{1,1},{2,1}},{{1,0},{2,0},{1,1},{1,2}},{{0,1},{1,1},{2,1},{2,2}},{{1,0},{1,1},{0,2},{1,2}}},
    {{{2,0},{0,1},{1,1},{2,1}},{{1,0},{1,1},{1,2},{2,2}},{{0,1},{1,1},{2,1},{0,2}},{{0,0},{1,0},{1,1},{1,2}}}
  };
  uint16_t colors[7]={0xFFDE,0xFE6A,0xFD20,0xFC80,0xFBE0,0xFB20,0xFD8A};

  uint8_t takePiece(){
    if(bagPos>=7){
      for(uint8_t i=0;i<7;i++) bag[i]=i;
      for(int i=6;i>0;i--){int j=random(i+1);int8_t t=bag[i];bag[i]=bag[j];bag[j]=t;}
      bagPos=0;
    }
    return bag[bagPos++];
  }
  bool hit(int nx,int ny,int nr) {
    for(int i=0;i<4;i++){
      int xx=nx+shape[type][nr][i][0], yy=ny+shape[type][nr][i][1];
      if(xx<0||xx>=W||yy>=H) return true;
      if(yy>=0 && board[yy][xx]) return true;
    }
    return false;
  }
  void spawn(){
    type=nextType; nextType=takePiece(); rot=0; px=3; py=0;
    if(hit(px,py,rot)) over=true;
    lastFall=millis();
  }
  void reset(){
    memset(board,0,sizeof(board)); score=0; lines=0; level=1; over=false;
    bagPos=7; nextType=takePiece(); spawn();
    selectWasDown=false; selectLongDone=false; backWasDown=false; nextWasDown=false; comboWasDown=false;
    lastFall=millis();
  }
  void cell(int x,int y,uint16_t c){
    int xx=X+x*CELL, yy=Y+y*CELL;
    tft.fillRoundRect(xx,yy,CELL-1,CELL-1,1,c);
    tft.drawFastHLine(xx+1,yy+1,CELL-3,CREAM);
  }
  void draw(){
    tft.fillScreen(BG);
    tft.setTextSize(1);
    tft.setTextColor(ORANGE); tft.setCursor(4,4); tft.print("ARGUS");
    tft.setTextColor(CREAM); tft.setCursor(51,4); tft.print("TETRIS");
    tft.setTextColor(PEACH); tft.setCursor(104,4); tft.print("S:"); tft.print(score);
    tft.fillRoundRect(X-2,Y-2,W*CELL+4,H*CELL+4,3,PANEL);
    tft.fillRect(X,Y,W*CELL,H*CELL,BG);
    for(int y=0;y<H;y++){
      tft.drawFastHLine(X,Y+y*CELL,W*CELL,GRID);
      for(int x=0;x<W;x++) if(board[y][x]) cell(x,y,colors[board[y][x]-1]);
    }
    if(!over){
      for(int i=0;i<4;i++){
        int xx=px+shape[type][rot][i][0], yy=py+shape[type][rot][i][1];
        if(xx>=0&&xx<W&&yy>=0&&yy<H) cell(xx,yy,colors[type]);
      }
    }
    if(over){
      tft.fillRoundRect(17,48,126,32,5,PANEL);
      tft.drawRoundRect(17,48,126,32,5,ORANGE);
      tft.setTextColor(CREAM); tft.setCursor(48,58); tft.print("GAME OVER");
      tft.setTextColor(PEACH); tft.setCursor(51,69); tft.print("SCORE "); tft.print(score);
    }
  }
  void lock(){
    for(int i=0;i<4;i++){
      int xx=px+shape[type][rot][i][0], yy=py+shape[type][rot][i][1];
      if(yy>=0&&yy<H&&xx>=0&&xx<W) board[yy][xx]=type+1;
    }
    int cleared=0;
    for(int y=H-1;y>=0;y--){
      bool full=true; for(int x=0;x<W;x++) if(!board[y][x]) {full=false;break;}
      if(full){
        cleared++;
        for(int yy=y;yy>0;yy--) for(int x=0;x<W;x++) board[yy][x]=board[yy-1][x];
        for(int x=0;x<W;x++) board[0][x]=0;
        y++;
      }
    }
    if(cleared){
      const uint16_t pts[]={0,100,300,500,800};
      score+=(cleared<=4?pts[cleared]:800)*level;
      lines+=cleared; level=1+lines/8;
      beepSuccess();
    }
    spawn();
  }
  void hardDrop(){
    int dist=0;
    while(!hit(px,py+1,rot)){py++;dist++;}
    score+=(uint32_t)dist*2;
    lock(); draw();
  }
  void tick(bool entered){
    if(entered){reset();draw();}
    uint32_t now=millis();
    bool b=(digitalRead(BTN_BACK)==LOW), n=(digitalRead(BTN_NEXT)==LOW), sel=(digitalRead(BTN_SELECT)==LOW);
    // SELECT: short press rotates/restarts; hold for four seconds exits.
    if(sel && !selectWasDown){selectWasDown=true;selectDownAt=now;selectLongDone=false;}
    if(sel && selectWasDown && !selectLongDone && now-selectDownAt>=4000){
      selectLongDone=true; screen=MAIN; beepBack(); return;
    }
    if(!sel && selectWasDown){
      uint32_t held=now-selectDownAt; selectWasDown=false;
      if(!selectLongDone && held<4000){
        if(over){reset();draw();}
        else{
          int nr=(rot+1)%4;
          if(!hit(px,py,nr))rot=nr;
          else if(!hit(px-1,py,nr)){px--;rot=nr;}
          else if(!hit(px+1,py,nr)){px++;rot=nr;}
          draw();
        }
      }
    }
    if(over){backWasDown=b;nextWasDown=n;return;}
    // Simultaneously press LEFT+RIGHT for a hard drop (one action per hold).
    if(b&&n&&!comboWasDown){hardDrop();comboWasDown=true;}
    if(!b||!n) comboWasDown=false;
    if(!(b&&n)){
      if(b && (!backWasDown || now-lastHorizontal>=125)){
        if(!hit(px-1,py,rot))px--;
        lastHorizontal=now; draw();
      }
      if(n && (!nextWasDown || now-lastHorizontal>=125)){
        if(!hit(px+1,py,rot))px++;
        lastHorizontal=now; draw();
      }
    }
    backWasDown=b; nextWasDown=n;
    uint32_t delayMs=max((uint32_t)95,(uint32_t)520-(uint32_t)(level-1)*32);
    if(now-lastFall>=delayMs){
      lastFall=now;
      if(!hit(px,py+1,rot))py++;
      else lock();
      draw();
    }
  }
}

// ================= ARGUS LINK: ESP-NOW SENSOR REQUEST =================
struct ArgusLinkPacket {
  uint32_t magic;
  uint8_t type; // 1=request, 2=response
  float temperature;
  float humidity;
  uint8_t ok;
};
static constexpr uint32_t ARGUS_LINK_MAGIC = 0xA612C3D1;
static const uint8_t ARGUS_LINK_BROADCAST[6] = {0xFF,0xFF,0xFF,0xFF,0xFF,0xFF};
volatile bool argusLinkResponseReady = false;
volatile bool argusLinkSensorOK = false;
volatile float argusLinkTemperature = 0.0f;
volatile float argusLinkHumidity = 0.0f;
volatile uint32_t argusLinkResponseAt = 0;
bool argusLinkStarted = false;
bool argusLinkWaiting = false;
uint32_t argusLinkRequestAt = 0;
uint32_t argusLinkLastResponseAt = 0;

void argusLinkOnRecv(const esp_now_recv_info_t *info, const uint8_t *data, int len) {
  if (!info || !data || len != (int)sizeof(ArgusLinkPacket)) return;
  ArgusLinkPacket packet;
  memcpy(&packet, data, sizeof(packet));
  if (packet.magic != ARGUS_LINK_MAGIC || packet.type != 2) return;
  argusLinkTemperature = packet.temperature;
  argusLinkHumidity = packet.humidity;
  argusLinkSensorOK = packet.ok != 0;
  argusLinkResponseAt = millis();
  argusLinkResponseReady = true;
}

bool argusLinkStart() {
  if (argusLinkStarted) return true;
  WiFi.mode(WIFI_STA);
  WiFi.disconnect(false, false);
  esp_wifi_set_channel(1, WIFI_SECOND_CHAN_NONE);
  if (esp_now_init() != ESP_OK) return false;
  esp_now_register_recv_cb(argusLinkOnRecv);
  esp_now_peer_info_t peer = {};
  memcpy(peer.peer_addr, ARGUS_LINK_BROADCAST, 6);
  peer.channel = 1;
  peer.encrypt = false;
  if (!esp_now_is_peer_exist(ARGUS_LINK_BROADCAST)) esp_now_add_peer(&peer);
  argusLinkStarted = true;
  return true;
}

void argusLinkStop() {
  if (argusLinkStarted) {
    esp_now_deinit();
    argusLinkStarted = false;
  }
  argusLinkWaiting = false;
}

void argusLinkRequest() {
  if (!argusLinkStart()) {
    argusLinkWaiting = false;
    return;
  }
  ArgusLinkPacket packet = {ARGUS_LINK_MAGIC, 1, 0.0f, 0.0f, 0};
  argusLinkResponseReady = false;
  argusLinkWaiting = true;
  argusLinkRequestAt = millis();
  esp_now_send(ARGUS_LINK_BROADCAST, (const uint8_t *)&packet, sizeof(packet));
}

void drawArgusLink() {
  clearScreen(COL_BG);
  tft.drawFastHLine(0, 22, 160, COL_ACCENT);
  tft.setTextColor(COL_ACCENT); tft.setTextSize(2); tft.setCursor(7, 4); tft.print("ARGUS LINK");
  tft.setTextSize(1);
  tft.drawRoundRect(6, 30, 148, 81, 7, COL_ACCENT2);
  tft.setTextColor(COL_DIM); tft.setCursor(15, 39); tft.print("TEMPERATURE");
  tft.setTextColor(COL_TEXT); tft.setTextSize(2); tft.setCursor(15, 52);
  if (argusLinkLastResponseAt == 0) tft.print("--.- C");
  else { tft.print((float)argusLinkTemperature, 1); tft.print(" C"); }
  tft.setTextSize(1); tft.setTextColor(COL_DIM); tft.setCursor(15, 79); tft.print("HUMIDITY");
  tft.setTextColor(COL_TEXT); tft.setTextSize(2); tft.setCursor(15, 91);
  if (argusLinkLastResponseAt == 0) tft.print("-- %");
  else { tft.print((int)(argusLinkHumidity + 0.5f)); tft.print(" %"); }
  tft.setTextSize(1); tft.setCursor(8, 119);
  if (!argusLinkStarted) { tft.setTextColor(COL_WARN); tft.print("LINK INIT ERROR"); }
  else if (argusLinkWaiting) { tft.setTextColor(COL_ACCENT2); tft.print("REQUESTING..."); }
  else if (argusLinkLastResponseAt == 0) { tft.setTextColor(COL_DIM); tft.print("NO DATA"); }
  else if (millis() - argusLinkLastResponseAt > 15000) { tft.setTextColor(COL_WARN); tft.print("DISCONNECTED"); }
  else if (!argusLinkSensorOK) { tft.setTextColor(COL_WARN); tft.print("SENSOR ERROR"); }
  else { tft.setTextColor(COL_ACCENT2); tft.print("CONNECTED"); }
  // No grey button hints: only a clean orange action button.
  tft.drawRoundRect(35, 136, 90, 20, 5, COL_ACCENT);
  tft.setTextColor(COL_ACCENT2); tft.setCursor(48, 143); tft.print("REQUEST");
}

void tickArgusLink(bool entered) {
  if (entered) {
    argusLinkWaiting = false;
    argusLinkResponseReady = false;
    if (!argusLinkStart()) { argusLinkStarted = false; }
    drawArgusLink();
  }
  if (backPressed()) { argusLinkStop(); screen = MAIN; return; }
  if (selectPressed()) {
    argusLinkRequest();
    drawArgusLink();
  }
  if (argusLinkResponseReady) {
    noInterrupts();
    argusLinkResponseReady = false;
    interrupts();
    argusLinkLastResponseAt = argusLinkResponseAt;
    argusLinkWaiting = false;
    drawArgusLink();
  }
  if (argusLinkWaiting && millis() - argusLinkRequestAt > 2500) {
    argusLinkWaiting = false;
    drawArgusLink();
  }
}

struct KarmaTarget { char ssid[32]; uint8_t channel; unsigned long lastSeen; bool seen; };

void IRAM_ATTR karmaCallback(void* buf, wifi_promiscuous_pkt_type_t type) {
if (type != WIFI_PKT_MGMT) return;
wifi_promiscuous_pkt_t* pkt = (wifi_promiscuous_pkt_t*)buf;
uint8_t* p = pkt->payload; int len = pkt->rx_ctrl.sig_len;
if (len < 24 || (p[0] & 0xF0) != 0x40) return;
karmaProbes++;
char tmp[32] = "";
for (int i = 24; i < len - 2; ) {
uint8_t tid = p[i], tl = p[i + 1];

if (tid == 0x00) {

memcpy(tmp, p + i + 2, l); tmp[l] = '\0'; break;
}
i += 2 + tl;
}
if (tmp[0] == '\0') return;
strncpy((char*)karmaLastSSID, tmp, 31); karmaLastSSID[31] = '\0';
}

// ================= SD CARD / SAVED IR CODES =================
// The SD module and the TFT share one SPI bus (SCK=12, MOSI=11, MISO=21).
// Both Adafruit_ST7735 and the SD library wrap their transfers in
// SPI.beginTransaction()/endTransaction() and drive their own CS pin, so
// reusing the already-initialized global `SPI` object (the same one the
// TFT uses, started in setup() with SPI.begin(12,21,11,-1)) is the safe,
// standard way to add a second device to that bus: both CS lines simply
// need to default HIGH so neither device is selected until a transfer is
// in progress for it. No second SPI host object is created, so the TFT's
// clock settings and behaviour are untouched.
bool sdReady = false;
const char* SD_IR_FILE = "/ARGUS_IR.txt";

struct SavedIRCommand {
  char name[16];
  int16_t protocol; // raw decode_type_t from IRrecv/IRsend, cast to int16_t
  uint64_t code;
  uint16_t bits;
  uint16_t repeat;
};
const int IR_SAVED_MAX = 24;
SavedIRCommand irSavedList[IR_SAVED_MAX];

bool sdInit() {
  pinMode(SD_CS, OUTPUT);
  digitalWrite(SD_CS, HIGH);   // SD deselected
  digitalWrite(TFT_CS, HIGH);  // TFT deselected before touching the shared bus
  bool ok = SD.begin(SD_CS, SPI, 1000000);
  if (ok && SD.cardType() == CARD_NONE) ok = false;
  digitalWrite(SD_CS, HIGH);   // release the bus cleanly for the TFT
  sdReady = ok;
  Serial.println(sdReady ? "SD INIT OK" : "SD INIT FAILED");
  return sdReady;
}

// Reads /ARGUS_IR.txt into irSavedList. Never truncates or rewrites the
// file. Malformed or incomplete lines are skipped, not fatal.
void sdLoadSavedIR() {
  irSavedCount = 0;
  if (!sdReady) return;
  File f = SD.open(SD_IR_FILE, FILE_READ);
  if (!f) return;
  while (f.available() && irSavedCount < IR_SAVED_MAX) {
    String line = f.readStringUntil('\n');
    line.trim();
    if (line.length() == 0) continue;
    int p1 = line.indexOf('|'); if (p1 < 0) continue;
    int p2 = line.indexOf('|', p1 + 1); if (p2 < 0) continue;
    int p3 = line.indexOf('|', p2 + 1); if (p3 < 0) continue;
    int p4 = line.indexOf('|', p3 + 1); if (p4 < 0) continue;
    SavedIRCommand& rec = irSavedList[irSavedCount];
    String name = line.substring(0, p1);
    strncpy(rec.name, name.c_str(), sizeof(rec.name) - 1);
    rec.name[sizeof(rec.name) - 1] = '\0';
    rec.protocol = (int16_t) line.substring(p1 + 1, p2).toInt();
    rec.code = strtoull(line.substring(p2 + 1, p3).c_str(), nullptr, 16);
    rec.bits = (uint16_t) line.substring(p3 + 1, p4).toInt();
    rec.repeat = (uint16_t) line.substring(p4 + 1).toInt();
    irSavedCount++;
  }
  f.close();
}

// Appends one line; existing content is never cleared or rewritten.
bool sdAppendSavedIR(const SavedIRCommand& rec) {
  if (!sdReady) return false;
  File f = SD.open(SD_IR_FILE, FILE_APPEND);
  if (!f) return false;
  char line[96];
  snprintf(line, sizeof(line), "%s|%d|%llX|%u|%u\n",
           rec.name, (int)rec.protocol,
           (unsigned long long)rec.code, rec.bits, rec.repeat);
  size_t n = f.print(line);
  f.close();
  return n > 0;
}

// Generic dispatch covers every protocol IRsend knows, not just the
// seven-entry IRProtocol enum used by the built-in TV database above.
bool sendSavedIR(const SavedIRCommand& rec) {
  uint32_t now = millis();
  if ((uint32_t)(now - lastIrSend) < IR_SEND_GUARD_MS) return false;
  irsend.send((decode_type_t)rec.protocol, rec.code, rec.bits, rec.repeat);
  lastIrSend = millis();
  playIRCue();
  return true;
}

irLastReceive = millis();
irrecv.resume();
drawIrReceiveScreen();
}
if (backPressed()) { irRecvStop(); screen = IR_MENU; return; }
if (selectPressed() && irResults.decode_type != UNKNOWN) {
bool ok = false;
if (sdReady && irSavedCount < IR_SAVED_MAX) {
SavedIRCommand rec;
snprintf(rec.name, sizeof(rec.name), "SAVED %d", irSavedCount + 1);
rec.protocol = (int16_t) irResults.decode_type;
rec.code = (uint64_t) irResults.value;
rec.bits = irResults.bits;
rec.repeat = 0;
ok = sdAppendSavedIR(rec);
if (ok) { irSavedList[irSavedCount] = rec; irSavedCount++; }
}
if (ok) { beepSuccess(); message("IR Receive", "Saved to SD", nullptr, COL_OK); }
else if (!sdReady) { beepError(); message("IR Receive", "SD not ready", "code not saved", COL_WARN); }
else if (irSavedCount >= IR_SAVED_MAX) { beepError(); message("IR Receive", "Saved list full", nullptr, COL_WARN); }
else { beepError(); message("IR Receive", "SD write failed", nullptr, COL_WARN); }
delay(600);
screen = IR_MENU;
}
if (millis() - irLastReceive > 6000) drawIrReceiveScreen();
}

// ================= IR SAVED =================
void drawIrSaved() {
clearScreen(); header("Saved IR", 1);
if (!sdReady) {
centerText("SD card required", 55, COL_WARN);
centerText("for saved codes", 75, COL_DIM);
centerText("BACK: return", 105, COL_DIM);
return;
}
if (irSavedCount == 0) {
centerText("No saved codes", 50, COL_ACCENT2);
centerText("Receive an IR code", 70, COL_DIM);
centerText("then SELECT to save", 82, COL_DIM);
centerText("BACK: return", 108, COL_DIM);
return;
}
const int visible = 5;
int first = irSavedIdx - visible / 2;
if (first < 0) first = 0;
if (irSavedCount <= visible) first = 0;
else if (first > irSavedCount - visible) first = irSavedCount - visible;
for (int row = 0; row < visible; row++) {
int idx = first + row; if (idx >= irSavedCount) break;
menuItem(32 + row * 16, irSavedList[idx].name, idx == irSavedIdx);
}
const SavedIRCommand& sel = irSavedList[irSavedIdx];
String proto = typeToString((decode_type_t)sel.protocol, false);
char info[32]; snprintf(info, sizeof(info), "%s  %u bit", proto.c_str(), sel.bits);
text(info, 4, 112, COL_ACCENT2);
char pos[12]; snprintf(pos, sizeof(pos), "%d/%d", irSavedIdx + 1, irSavedCount);
text(pos, 128, 120, COL_DIM);
text("NEXT", 4, 120, COL_DIM);
text("SELECT SEND", 38, 120, COL_DIM);
if (irToastUntil > millis()) {
tft.fillRoundRect(20, 45, 120, 38, 5, COL_OK);
text("SENT", 61, 51, COL_BG, 2);
}
}
void tickIrSaved(bool entered) {
if (entered) { sdLoadSavedIR(); irSavedIdx = 0; drawIrSaved(); }
if (backPressed()) { screen = IR_MENU; return; }
if (!sdReady || irSavedCount == 0) return;
if (nextPressed()) { irSavedIdx = (irSavedIdx + 1) % irSavedCount; drawIrSaved(); }
if (selectPressed()) {
if (sendSavedIR(irSavedList[irSavedIdx])) { irToastUntil = millis() + 700; drawIrSaved(); }
}
if (irToastUntil != 0 && millis() >= irToastUntil) { irToastUntil = 0; drawIrSaved(); }
}

// ================= SETUP & LOOP =================
void setup() {
Serial.begin(115200);
delay(200);

rgbLed.begin();
rgbLed.setBrightness(70);
rgbLed.clear();
rgbLed.show();

pinMode(BTN_BACK, INPUT_PULLUP);
pinMode(BTN_NEXT, INPUT_PULLUP);
pinMode(BTN_SELECT, INPUT_PULLUP);
pinMode(TFT_BL, OUTPUT);
pinMode(TFT_CS, OUTPUT);
pinMode(SD_CS, OUTPUT);
pinMode(BUZZER_PIN, OUTPUT);
pinMode(IR_PIN, OUTPUT);
digitalWrite(TFT_CS, HIGH);
digitalWrite(SD_CS, HIGH);
digitalWrite(IR_PIN, LOW);

SPI.begin(12, 21, 11, -1);

ledcAttach(BUZZER_PIN, 2000, 10);

tft.initR(INITR_BLACKTAB);
tft.setRotation(1); // Flipped to the opposite landscape orientation
tft.setTextWrap(false);
applyBrightness();

clearScreen(COL_BG);
// Clean ARGUS boot screen: no eye animation, orange monochrome only.
tft.fillScreen(COL_BG);
tft.setTextColor(COL_ACCENT);
tft.setTextSize(3);
tft.setCursor(22, 36);
tft.print("ARGUS");
tft.setTextSize(1);
tft.setTextColor(COL_ACCENT2);
tft.setCursor(56, 66);
tft.print("SYSTEM v2.0");
tft.drawRoundRect(12, 94, 136, 8, 3, COL_ACCENT);
for (int progress = 0; progress <= 128; progress += 8) {
  tft.fillRoundRect(16, 97, progress, 2, 1, COL_ACCENT2);
  delay(18);
}
delay(120);

// SD init happens after the TFT is fully up and done drawing, so the
// shared SPI bus is never touched by both devices at once. A missing or
// failed card is reported in Serial and on the INFO screen; nothing here
// blocks or loops waiting for a card.
sdInit();

irsend.begin();
irrecv.enableIRIn();
irrecv.disableIRIn();

WiFi.mode(WIFI_STA);
WiFi.disconnect();

randomSeed(micros());
playBootMelody();
delay(500);

screen = MAIN;
prevScreen = MAIN;
screenEntered = true;
}

void loop() {
bool entered = screenEntered;
screenEntered = false;

switch (screen) {
case MAIN:              tickMain(entered);              break;
case WIFI_MENU:         tickWifiMenu(entered);          break;
case WIFI_SCAN:         tickWifiScan(entered);          break;
case WIFI_INFO:         tickWifiInfo(entered);          break;
case WIFI_STATUS:       tickWifiStatus(entered);        break;
case WIFI_HOTSPOT:      tickHotspot(entered);           break;
case BEACON_SPAM:       tickBeaconSpam(entered);        break;
case CAPTIVE_PORTAL:    tickCaptivePortal(entered);     break;
case SNIFFER:           tickSniffer(entered);           break;
case DEAUTH:            tickDeauth(entered);            break;
case RADAR:             tickRadar(entered);             break;
case DEAUTH_DETECTOR:   tickDeauthDetector(entered);    break;
case KARMA:             tickKarma(entered);             break;
case IR_MENU:           tickIrMenu(entered);            break;
case IR_TV_BRANDS:      tickIrBrands(entered);          break;
case IR_TV_MODELS:      tickIrModels(entered);          break;
case IR_TV_REMOTE:      tickIrRemote(entered);          break;
case IR_RECEIVE:        tickIrReceive(entered);         break;
case IR_SAVED:          tickIrSaved(entered);           break;
case DISPLAY_TEST:      tickDisplayTest(entered);       break;
case INFO:              tickInfo(entered);              break;
case SETTINGS:          tickSettings(entered);          break;
case ARGUS_HUB:         tickArgusHub(entered);           break;
case RGB_MENU:          tickRGBMenu(entered);              break;
 case TETRIS_GAME:       CuteTetris::tick(entered);              break;
case ARGUS_LINK:        tickArgusLink(entered);                   break;

default:                screen = MAIN;                  break;

}

if (screen != prevScreen) {
prevScreen = screen;
screenEntered = true;
}
rgbTick();
delay(5);
}
